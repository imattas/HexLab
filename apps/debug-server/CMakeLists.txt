add_executable(hexlab_debug_server main.cpp)
target_link_libraries(hexlab_debug_server PRIVATE hexlab::debugger)
if(WIN32)
  target_link_libraries(hexlab_debug_server PRIVATE ws2_32)
elseif(APPLE)
  set_target_properties(hexlab_debug_server PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
elseif(UNIX)
  set_target_properties(hexlab_debug_server PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
if(WIN32)
  install(TARGETS hexlab_debug_server
    RUNTIME_DEPENDENCY_SET hexlab_windows_runtime_dependencies
    RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
else()
  install(TARGETS hexlab_debug_server RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
endif()

if(NOT COMMAND hexlab_add_library)
  cmake_minimum_required(VERSION 3.24)
  project(HexLabSymbols VERSION 0.1.0 LANGUAGES CXX)
  include(GNUInstallDirs)
  include(GenerateExportHeader)
  include(CTest)
  include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/HexLabLibrary.cmake")
  add_library(hexlab_options INTERFACE)
  target_compile_features(hexlab_options INTERFACE cxx_std_20)
  add_library(hexlab_types STATIC "${CMAKE_CURRENT_SOURCE_DIR}/../types/src/types.cpp")
  add_library(hexlab::types ALIAS hexlab_types)
  generate_export_header(hexlab_types EXPORT_FILE_NAME "${PROJECT_BINARY_DIR}/generated/hexlab/types/types_export.hpp")
  target_include_directories(hexlab_types PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/../types/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/../core/include"
    "${PROJECT_BINARY_DIR}/generated")
  target_link_libraries(hexlab_types PUBLIC hexlab_options)
endif()

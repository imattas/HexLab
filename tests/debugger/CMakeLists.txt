if(NOT TARGET hexlab::debugger)
  set(HEXLAB_DEBUGGER_STANDALONE TRUE)
  cmake_minimum_required(VERSION 3.24)
  project(HexLabDebuggerTests VERSION 0.1.0 LANGUAGES C CXX)
  include(CTest)
  include(GNUInstallDirs)
  include(GenerateExportHeader)
  include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/HexLabLibrary.cmake")
  add_library(hexlab_options INTERFACE)
  target_compile_features(hexlab_options INTERFACE cxx_std_20)
  target_compile_options(hexlab_options INTERFACE
    $<$<COMPILE_LANG_AND_ID:CXX,MSVC>:/W4;/permissive-;/utf-8>
    $<$<COMPILE_LANG_AND_ID:CXX,GNU,Clang,AppleClang>:-Wall;-Wextra;-Wpedantic>)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../libs/core" "${CMAKE_CURRENT_BINARY_DIR}/hexlab-core")
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../libs/debugger" "${CMAKE_CURRENT_BINARY_DIR}/hexlab-debugger")
endif()
add_executable(hexlab_debugger_tests debugger_tests.cpp)
target_link_libraries(hexlab_debugger_tests PRIVATE hexlab::debugger)
target_include_directories(hexlab_debugger_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}/..")
if(HEXLAB_DEBUGGER_STANDALONE)
  add_custom_command(TARGET hexlab_debugger_tests POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:hexlab_core>" "$<TARGET_FILE_DIR:hexlab_debugger_tests>"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:hexlab_debugger>" "$<TARGET_FILE_DIR:hexlab_debugger_tests>")
endif()
set(_hexlab_debugger_test_args)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  include(CheckCXXSourceCompiles)
  set(_hexlab_saved_required_flags "${CMAKE_REQUIRED_FLAGS}")
  set(_hexlab_saved_required_link_options "${CMAKE_REQUIRED_LINK_OPTIONS}")
  set(CMAKE_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS} -m32")
  set(CMAKE_REQUIRED_LINK_OPTIONS "${CMAKE_REQUIRED_LINK_OPTIONS};-m32")
  check_cxx_source_compiles("int main() { return 0; }" HEXLAB_CAN_BUILD_I386_DEBUGGEE)
  set(CMAKE_REQUIRED_FLAGS "${_hexlab_saved_required_flags}")
  set(CMAKE_REQUIRED_LINK_OPTIONS "${_hexlab_saved_required_link_options}")
  unset(_hexlab_saved_required_flags)
  unset(_hexlab_saved_required_link_options)
  if(HEXLAB_CAN_BUILD_I386_DEBUGGEE)
    enable_language(ASM)
    add_executable(hexlab_debugger_i386_fixture i386_debuggee.S)
    target_compile_options(hexlab_debugger_i386_fixture PRIVATE -m32 -fno-pie)
    target_link_options(hexlab_debugger_i386_fixture PRIVATE -m32 -nostdlib -static -no-pie)
    list(APPEND _hexlab_debugger_test_args "$<TARGET_FILE:hexlab_debugger_i386_fixture>")
  endif()
endif()
add_test(NAME debugger COMMAND hexlab_debugger_tests ${_hexlab_debugger_test_args})
set_tests_properties(debugger PROPERTIES TIMEOUT 30)
add_executable(hexlab_debugger_remote_tests ../debugger_remote_tests.cpp)
target_link_libraries(hexlab_debugger_remote_tests PRIVATE hexlab::debugger)
if(WIN32)
  target_link_libraries(hexlab_debugger_remote_tests PRIVATE ws2_32)
endif()
if(HEXLAB_DEBUGGER_STANDALONE)
  add_custom_command(TARGET hexlab_debugger_remote_tests POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:hexlab_core>" "$<TARGET_FILE_DIR:hexlab_debugger_remote_tests>"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:hexlab_debugger>" "$<TARGET_FILE_DIR:hexlab_debugger_remote_tests>")
endif()
add_test(NAME debugger_remote COMMAND hexlab_debugger_remote_tests)
set_tests_properties(debugger_remote PROPERTIES TIMEOUT 30)
